Modelado de Datos — Prisma Schema

El esquema extiende el Product actual con campos de costo y umbrales, agrega trazabilidad de lotes (ProductLot), y formaliza el concepto de Combo mixto (ServiceCombo). El modelo ServiceIngredient ya existente actúa como la BOM principal.

1.1 Modelo Product — campos nuevos

🔑
Campos críticos a agregar Los campos de costo (costPrice) y umbrales (minStock, alertThreshold) son la base de todo el módulo de rentabilidad. Sin ellos no hay COGS.

model Product {

}

enum LotStrategy {
  FEFO  // First Expired, First Out — recomendado para cosméticos
  FIFO  // First In, First Out

1.2 Modelo ProductLot — trazabilidad de lotes

⚗️
¿Por qué lotes en un salón? Los cosméticos tienen fecha de caducidad. Un esmalte gel o una ampolla de keratina abierta hace 6 meses puede dañar a una clienta. La estrategia FEFO (expiración primero) descuenta siempre el lote más próximo a vencer.
schema.prisma — ProductLot (NUEVO)
model ProductLot {
  id          String    @id @db.Uuid
  tenantId    String    @db.Uuid
  productId   String    @db.Uuid
  lotNumber   String                    // número de lote del proveedor
  expiresAt   DateTime? @db.Date        // fecha de caducidad (null = no aplica)
  quantity    Decimal   @db.Decimal(10, 3)  // stock actual de este lote
  initialQty  Decimal   @db.Decimal(10, 3)  // cantidad original al ingresar
  costPrice   Decimal?  @db.Decimal(10, 4)  // costo de este lote específico
  receivedAt  DateTime  @default(now()) @db.Timestamptz
  createdAt   DateTime  @default(now()) @db.Timestamptz

  tenant   Tenant     @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  product  Product    @relation(fields: [productId], references: [id], onDelete: Cascade)

  // Índice crítico para FEFO: ordenar por expiresAt ASC

2.1 Descuento automático de inventario por cita

⚡
Atomicidad obligatoria Todo el descuento de BOM ocurre dentro de una sola transacción Prisma ($transaction). Si falla el descuento de cualquier ingrediente, se revierte todo. Nunca stock parcialmente descontado.
inventory.service.ts — deductBomForAppointment()
import { Injectable, BadRequestException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { uuidv7 } from 'uuidv7';
import Decimal from 'decimal.js';

@Injectable()
export class InventoryService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * Descuenta automáticamente todos los ingredientes del BOM del servicio.
   * Soporta conversión de unidades (ml ← unidades con unitQty).
   * Aplica FEFO/FIFO si el producto tiene trazabilidad de lotes.
   * Registra StockMovement con COGS calculado.
   */
  async deductBomForAppointment(
    appointmentId: string,
    serviceId: string,
    tenantId: string,
  ): Promise<void> {
    // 1. Obtener BOM del servicio con info de producto
    const ingredients = await (this.prisma as any).serviceIngredient.findMany({
      where: { serviceId, tenantId },
      include: {
        product: {
          include: { lots: { where: { quantity: { gt: 0 } }, orderBy: [{ expiresAt: 'asc' }, { receivedAt: 'asc' }] } },
        },
      },
    });

    if (!ingredients.length) return; // servicio sin BOM, nada que descontar

    // 2. Validar stock suficiente antes de iniciar transacción
    for (const ing of ingredients) {
      const product = ing.product;
      const required = new Decimal(ing.quantity);

      // Convertir: si el recipe usa ml y el stock está en unidades
      const stockInRecipeUnit = product.unit === 'unit' && product.unitQty
        ? new Decimal(product.stock).mul(product.unitQty)
        : new Decimal(product.stock);

      if (stockInRecipeUnit.lt(required)) {
        throw new BadRequestException(
          `Stock insuficiente para "${product.name}": necesitas ${required} ${ing.unit}, hay ${stockInRecipeUnit.toFixed(3)} ${ing.unit}`
        );
      }
    }

    // 3. Transacción atómica: descontar todo o nada
    await (this.prisma as any).$transaction(async (tx: any) => {
      for (const ing of ingredients) {
        const product = ing.product;
        let remaining = new Decimal(ing.quantity); // cantidad a descontar (en recipe unit)
        let totalCogs = new Decimal(0);
        const lotMovements: { lotId: string; quantity: Decimal }[] = [];

        if (product.hasLotTracking && product.lots.length) {
          // FEFO: descontar de lotes ordenados por expiresAt ASC
          for (const lot of product.lots) {
            if (remaining.lte(0)) break;

            // Convertir stock de lote a recipe unit
            const lotQtyInRecipeUnit = product.unitQty
              ? new Decimal(lot.quantity).mul(product.unitQty)
              : new Decimal(lot.quantity);

            const takeFromLot = Decimal.min(remaining, lotQtyInRecipeUnit);
            const takeInStockUnit = product.unitQty
              ? takeFromLot.div(product.unitQty)
              : takeFromLot;

            // Descontar del lote
            await tx.productLot.update({
              where: { id: lot.id },
              data: { quantity: { decrement: takeInStockUnit.toNumber() } },
            });

            const lotCost = lot.costPrice
              ? takeInStockUnit.mul(lot.costPrice)
              : new Decimal(0);
            totalCogs = totalCogs.add(lotCost);
            lotMovements.push({ lotId: lot.id, quantity: takeInStockUnit });
            remaining = remaining.minus(takeFromLot);
          }
        }

        // Descontar del stock total del producto
        const deductInStockUnit = product.unitQty
          ? new Decimal(ing.quantity).div(product.unitQty)
          : new Decimal(ing.quantity);

        const updatedProduct = await tx.product.update({
          where: { id: product.id },
          data: { stock: { decrement: deductInStockUnit.toNumber() } },
        });

        // Si no hay lotes: calcular COGS por precio del producto
        if (!product.hasLotTracking && product.costPrice) {
          totalCogs = deductInStockUnit.mul(product.costPrice);
        }

        // Registrar StockMovement
        const movementId = uuidv7();
        await tx.stockMovement.create({
          data: {
            id: movementId,
            tenantId,
            productId: product.id,
            appointmentId,
            reason: 'appointment_use',
            delta: -deductInStockUnit.toNumber(),
            stockAfter: updatedProduct.stock,
            unit: product.unit,
            cogsCost: totalCogs.gt(0) ? totalCogs.toNumber() : null,
            notes: `BOM: ${ing.notes ?? ing.unit}`,
            // Detalle de lotes descontados
            lotMovements: lotMovements.length ? {
              create: lotMovements.map(lm => ({
                id: uuidv7(), lotId: lm.lotId, quantity: lm.quantity.toNumber(),
              })),
            } : undefined,
          },
        });
      }
    });
  }
}

2.2 Alerta Predictiva de Desabastecimiento

📡
La diferencia vs. un sistema normal Un ERP básico avisa cuando ya te quedaste sin stock. Este módulo avisa antes: "el miércoles tienes 6 citas de keratina y solo hay 4 ampollas".
inventory.service.ts — checkPredictiveAlerts()
/**
 * Calcula cuántos de cada producto se consumirán en los próximos N días
 * basándose en citas ya agendadas. Si el stock proyectado no alcanza,
 * retorna una alerta con el día en que se agotará y cuánto falta pedir.
 */
async checkPredictiveAlerts(
  tenantId: string,
  horizonDays = 7,
): Promise<PredictiveAlert[]> {
  const horizon = new Date();
  horizon.setDate(horizon.getDate() + horizonDays);

  // 1. Citas pendientes/confirmadas en el horizonte
  const appointments = await this.prisma.appointment.findMany({
    where: {
      tenantId,
      status: { in: ['pending', 'confirmed_by_ai', 'attending'] },
      startsAt: { gte: new Date(), lte: horizon },
      serviceId: { not: null },
    },
    select: { serviceId: true, startsAt: true },
  });

  if (!appointments.length) return [];

  // 2. Agrupar consumo proyectado por producto
  const serviceIds = [...new Set(appointments.map(a => a.serviceId!))];
  const allIngredients = await (this.prisma as any).serviceIngredient.findMany({
    where: { serviceId: { in: serviceIds }, tenantId },
    include: { product: true },
  });

  // productId → { product, projectedConsumption, consumingDays }
  const projectionMap = new Map<string, {
    product: any;
    projected: Decimal;
    consumingDates: Date[];
  }>();

  for (const appt of appointments) {
    const ings = allIngredients.filter(i => i.serviceId === appt.serviceId);
    for (const ing of ings) {
      const entry = projectionMap.get(ing.productId) ?? {
        product: ing.product,
        projected: new Decimal(0),
        consumingDates: [],
      };

      // Convertir a unidades de stock si hay factor unitQty
      const qtyInStockUnit = ing.product.unitQty
        ? new Decimal(ing.quantity).div(ing.product.unitQty)
        : new Decimal(ing.quantity);

      entry.projected = entry.projected.add(qtyInStockUnit);
      entry.consumingDates.push(new Date(appt.startsAt));
      projectionMap.set(ing.productId, entry);
    }
  }

  // 3. Comparar con stock actual y generar alertas
  const alerts: PredictiveAlert[] = [];

  for (const [productId, entry] of projectionMap) {
    const currentStock = new Decimal(entry.product.stock);
    const shortage = entry.projected.minus(currentStock);

    if (shortage.gt(0)) {
      // Calcular el día en que se agotará (acumulando consumo por fecha)
      const stockoutDate = this.estimateStockoutDate(
        currentStock, entry.consumingDates, allIngredients, productId,
      );

      alerts.push({
        productId,
        productName: entry.product.name,
        currentStock: currentStock.toNumber(),
        projectedConsumption: entry.projected.toNumber(),
        shortage: shortage.toNumber(),
        unit: entry.product.unit,
        stockoutDate,
        urgency: shortage.gte(currentStock.mul(2)) ? 'critical' : 'warning',
        // Mensaje en lenguaje llano para la dueña
        humanMessage: this.buildHumanMessage(entry.product, entry.projected, currentStock, stockoutDate),
      });
    }
  }

  return alerts.sort((a, b) => a.urgency === 'critical' ? -1 : 1);
}

private buildHumanMessage(product: any, projected: Decimal, stock: Decimal, date: Date): string {
  const dow = ['domingo', 'lunes', 'martes', 'miércoles', 'jueves', 'viernes', 'sábado'][date.getDay()];
  return `⚠️ Se te agotarán las "${product.name}" el ${dow}: necesitas ${projected.toFixed(1)} ${product.unit} y solo tienes ${stock.toFixed(1)}.`;
}

2.3 Cálculo de COGS y Margen de Utilidad Real

💰
Por qué esto cambia el negocio Un servicio de "Baño de Color" puede cobrar $45 pero consumir $38 en productos. Sin este cálculo la dueña piensa que gana bien. Con él, ve que está ganando $7 — y puede ajustar precios o reducir consumo.
inventory.service.ts — getServiceCOGS() y getServiceMargin()
interface CogBreakdown {
  ingredientName: string;
  quantity: number;
  unit: string;
  unitCost: number;  // costo por unidad de stock
  lineTotal: number; // cantidad × costo
}
interface ServiceMargin {
  serviceId: string;
  serviceName: string;
  basePrice: number;
  totalCogs: number;
  grossMargin: number;     // basePrice - totalCogs
  marginPercent: number;   // (grossMargin / basePrice) × 100
  breakdown: CogBreakdown[];
  rating: 'excellent' | 'good' | 'tight' | 'loss';
}

async getServiceMargin(serviceId: string, tenantId: string): Promise<ServiceMargin> {
  const [service, ingredients] = await Promise.all([
    this.prisma.service.findFirstOrThrow({ where: { id: serviceId, tenantId } }),
    (this.prisma as any).serviceIngredient.findMany({
      where: { serviceId, tenantId },
      include: { product: true },
    }),
  ]);

  const breakdown: CogBreakdown[] = [];
  let totalCogs = new Decimal(0);

  for (const ing of ingredients) {
    const product = ing.product;
    if (!product.costPrice) continue;

    // Si la receta usa ml y el stock está en unidades, convertir el costo
    let costPerRecipeUnit: Decimal;
    if (product.unit === 'unit' && product.unitQty && ing.unit !== 'unit') {
      // costPrice está en "por unidad". La receta usa ml.
      // Costo por ml = costPrice / unitQty
      costPerRecipeUnit = new Decimal(product.costPrice).div(product.unitQty);
    } else {
      costPerRecipeUnit = new Decimal(product.costPrice);
    }

    const lineTotal = costPerRecipeUnit.mul(ing.quantity);
    totalCogs = totalCogs.add(lineTotal);

    breakdown.push({
      ingredientName: product.name,
      quantity: Number(ing.quantity),
      unit: ing.unit,
      unitCost: costPerRecipeUnit.toNumber(),
      lineTotal: lineTotal.toNumber(),
    });
  }

  const basePrice = new Decimal(service.basePrice);
  const grossMargin = basePrice.minus(totalCogs);
  const marginPct = basePrice.gt(0) ? grossMargin.div(basePrice).mul(100).toNumber() : 0;

  return {
    serviceId, serviceName: service.name,
    basePrice: basePrice.toNumber(),
    totalCogs: totalCogs.toNumber(),
    grossMargin: grossMargin.toNumber(),
    marginPercent: marginPct,
    breakdown,
    rating: marginPct >= 60 ? 'excellent'
          : marginPct >= 35 ? 'good'
          : marginPct >= 10 ? 'tight'
          : 'loss',
  };
}

// Rentabilidad de todos los servicios del tenant (para dashboard)
async getAllServiceMargins(tenantId: string): Promise<ServiceMargin[]> {
  const services = await this.prisma.service.findMany({
    where: { tenantId, active: true },
    select: { id: true },
  });
  const margins = await Promise.all(
    services.map(s => this.getServiceMargin(s.id, tenantId))
  );
  return margins.sort((a, b) => a.marginPercent - b.marginPercent);
}

// BomBuilder — armado visual de receta de servicio
// Props: serviceId, servicePrice, tenantProducts, onSave

interface BomBuilderProps {
  serviceId: string;
  servicePrice: number;
  tenantProducts: SalonProduct[];
  onSave: (ingredients: IngredientRow[]) => Promise<void>;
}

interface IngredientRow {
  productId: string;
  quantity: number;
  unit: string;   // ml | g | unid. — auto-set desde producto
  notes?: string;
  // computed client-side para display:
  productName: string;
  lineCost: number;   // quantity × costPerRecipeUnit
}

export function BomBuilder({ serviceId, servicePrice, tenantProducts, onSave }: BomBuilderProps) {
  const [rows, setRows] = useState<IngredientRow[]>([]);
  const [search, setSearch] = useState('');

  const totalCogs = rows.reduce((sum, r) => sum + r.lineCost, 0);
  const margin = servicePrice - totalCogs;
  const marginPct = servicePrice > 0 ? (margin / servicePrice) * 100 : 0;
  const ratingEmoji = marginPct >= 60 ? '🟢' : marginPct >= 35 ? '🔵' : marginPct >= 10 ? '🟡' : '🔴';

  function addProduct(product: SalonProduct) {
    const recipeUnit = product.unit === 'unit' && product.unitQtyUnit
      ? product.unitQtyUnit  // e.g. 'ml' si la botella es en ml
      : product.unit;
    const costPerRecipeUnit = product.costPrice && product.unitQty
      ? product.costPrice / product.unitQty
      : (product.costPrice ?? 0);
    setRows(prev => [...prev, {
      productId: product.id, productName: product.name,
      quantity: 1, unit: recipeUnit, lineCost: costPerRecipeUnit,
    }]);
  }

  function updateQty(idx: number, qty: number, product: SalonProduct) {
    const costPerRecipeUnit = product.costPrice && product.unitQty
      ? product.costPrice / product.unitQty : (product.costPrice ?? 0);
    setRows(prev => prev.map((r, i) => i === idx
      ? { ...r, quantity: qty, lineCost: costPerRecipeUnit * qty }
      : r
    ));
  }

  return (
    <div className="bom-builder">
      <ProductSearch products={tenantProducts} search={search} onSearch={setSearch} onAdd={addProduct} />
      <IngredientTable rows={rows} products={tenantProducts} onChangeQty={updateQty} onRemove={idx => setRows(p => p.filter((_,i) => i !== idx))} />
      <CogsSummary totalCogs={totalCogs} servicePrice={servicePrice} margin={margin} marginPct={marginPct} ratingEmoji={ratingEmoji} />
      <button onClick={() => onSave(rows)}>Guardar receta</button>
    </div>
  );
}

components/StockAlertPanel.tsx
interface StockAlertPanelProps {
  tenantId: string;
  horizonDays?: number; // default 7
}

export function StockAlertPanel({ tenantId, horizonDays = 7 }: StockAlertPanelProps) {
  const { data: alerts, isLoading } = useQuery({
    queryKey: ['stock-alerts', tenantId, horizonDays],
    queryFn: () => api.get<PredictiveAlert[]>(
      `/inventory/alerts?tenantId=${tenantId}&horizonDays=${horizonDays}`
    ),
    refetchInterval: 5 * 60_000, // refrescar cada 5 min
    staleTime: 2 * 60_000,
  });

  if (isLoading) return <AlertSkeleton />;
  const criticals = alerts?.filter(a => a.urgency === 'critical') ?? [];
  const warnings  = alerts?.filter(a => a.urgency === 'warning')  ?? [];

  return (
    <section className="stock-alert-panel">
      <h3>📦 Inventario · próximos {horizonDays} días</h3>
      {criticals.map(a => (
        <AlertCard key={a.productId} alert={a} variant="critical"
          message={a.humanMessage}
          cta="Pedir ahora"
          onCta={() => /* abrir orden de compra */}
        />
      ))}
      {warnings.map(a => (
        <AlertCard key={a.productId} alert={a} variant="warning"
          message={a.humanMessage}
        />
      ))}
      {!criticals.length && !warnings.length && (
        <AlertCard variant="ok"
          message="✅ Todo el stock está cubierto para esta semana."
        />
      )}
    </section>
  );
}

// Drawer de desglose de costos (CogsBomDrawer)
function CogsBomDrawer({ margin }: { margin: ServiceMargin }) {
  return (
    <Drawer title={`Desglose: ${margin.serviceName}`}>
      <div className="summary-row">
        <BigStat label="Precio" value={`$${margin.basePrice}`} />
        <BigStat label="Insumos" value={`-$${margin.totalCogs.toFixed(2)}`} color="rose" />
        <BigStat label="Ganancia" value={`$${margin.grossMargin.toFixed(2)}`} color="emerald" />
      </div>
      <p className="margin-label">
        {margin.ratingEmoji} Por cada cita de {margin.serviceName}, ganas en limpio 
        ${margin.grossMargin.toFixed(2)} ({margin.marginPercent.toFixed(0)}% de margen).
      </p>
      <h4>¿En qué se va el resto?</h4>
      {margin.breakdown.map(b => (
        <BomCostRow key={b.ingredientName}
          name={b.ingredientName}
          qty={b.quantity} unit={b.unit}
          lineCost={b.lineTotal}
        />
      ))}
    </Drawer>
  );
}
